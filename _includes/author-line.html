{% comment %} Author line for a publication (list entry, its own page, homepage). `authors` is written once, either - as one line of text exactly as the journal prints it: shown verbatim, with your name in bold wherever one of `site.publication_author_names` appears in it; or - as a list of "Surname, I." names: joined APA-style ("A, B, & C"), bolding entries that contain `site.publication_author`. Cite and the Google Scholar tags get individual names from _includes/author-parse.html either way. Usage: {% include author-line.html item=item %} {% endcomment %} {%- assign al_item = include.item -%} {%- if al_item.authors.first -%} {%- assign al_me = site.publication_author | default: "Marais, K" -%} {%- for author in al_item.authors -%}{%- if author contains al_me -%}{{ author }}{%- else -%}{{ author }}{%- endif -%}{%- unless forloop.last -%}{%- if forloop.rindex == 2 -%}, & {% else %}, {% endif -%}{%- endunless -%}{%- endfor -%} {%- else -%} {%- assign al_text = al_item.authors | append: "" | escape -%} {%- for name in site.publication_author_names -%} {%- capture al_token -%}@@AUTHOR{{ forloop.index }}@@{%- endcapture -%} {%- assign al_text = al_text | replace: name, al_token -%} {%- endfor -%} {%- for name in site.publication_author_names -%} {%- capture al_token -%}@@AUTHOR{{ forloop.index }}@@{%- endcapture -%} {%- capture al_bold -%}{{ name }}{%- endcapture -%} {%- assign al_text = al_text | replace: al_token, al_bold -%} {%- endfor -%} {{- al_text -}} {%- endif -%}
